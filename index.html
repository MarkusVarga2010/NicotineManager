create table eigene_schulden (
  id bigint generated always as identity primary key,
  user_id uuid not null,
  glaeubiger text not null,
  betrag numeric not null,
  notiz text,
  bezahlt boolean not null default false,
  created_at timestamptz not null default now(),
  bezahlt_am timestamptz
);
alter table eigene_schulden enable row level security;
create policy "eigene schulden eigener user"
  on eigene_schulden for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);

Lagerbestand
Trag hier einfach ein, was du von jedem Produkt gerade tatsächlich vorrätig hast. Produkte unter deiner Warnschwelle werden oben hervorgehoben.
Noch nicht eingerichtet
Dafür fehlt in Supabase noch eine Tabelle. SQL zum Anlegen:
create table lagerbestand_manuell (
  id bigint generated always as identity primary key,
  user_id uuid not null,
  produkt text not null,
  menge integer not null default 0,
  schwelle integer not null default 5,
  updated_at timestamptz not null default now(),
  unique(user_id, produkt)
);
alter table lagerbestand_manuell enable row level security;
create policy "lagerbestand manuell eigener user"
  on lagerbestand_manuell for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
Warnschwelle noch nicht verfügbar
Der Bestand selbst wird schon gespeichert — nur die Warnschwelle pro Produkt braucht noch eine zusätzliche Spalte. SQL zum Nachrüsten:
alter table lagerbestand_manuell
  add column if not exists schwelle integer not null default 5;

Eigenverbrauch
Trag hier ein, was du selbst konsumiert hast (nicht verkauft) — der Lagerbestand wird automatisch entsprechend reduziert.
Noch nicht eingerichtet
Dafür fehlt in Supabase noch eine Tabelle. SQL zum Anlegen:
create table eigenverbrauch (
  id bigint generated always as identity primary key,
  user_id uuid not null,
  produkt text not null,
  variante_id bigint,
  variante_name text,
  menge integer not null default 1,
  notiz text,
  erfasst_am timestamptz not null default now(),
  created_at timestamptz not null default now()
);
alter table eigenverbrauch enable row level security;
create policy "eigenverbrauch eigener user"
  on eigenverbrauch for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
